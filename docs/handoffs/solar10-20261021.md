## 交核心：共用技能施放橫幅誤用角色頭像（待修）

最小資料例即本包太陽：`{id:"solar10:character.sun",skill:"m1:armor",portrait:"",avatar:"",sprite:""}`（完整可校驗定義見extension）。以core＋本包，將太陽排前鋒進蔚藍前哨，施放防禦增幅。

實際：`src/presentation/ui.ts:794` 用 `characters.find(c => c.skill === sk.id)` 取第一個技能持有人，太陽橫幅出現火星頭像。手機截圖 `.artifacts/solar10/battle-390.png` 已人工開圖確認；其他共用技能同樣可能指向既有六人。角色卡／詳情／戰鬥sprite引用本身正確，戰力與冷卻也正確。

玩家預期：橫幅顯示實際施法者太陽的無圖備援（補圖後用太陽頭像），不可把火星當施法者。建議核心用Flash的可選施法單位ID／角色定義ID傳遞來源，UI依實際來源讀portrait；缺來源時可只顯示技能文字，不能猜第一位持有人。這是建議，實際通用API由核心決定。

相容：如Flash新增可選來源欄位，保留舊呼叫fallback；不改持久save／包schema／技能ID，不加角色ID特例。驗收：太陽、火星分別施放同一技能仍各顯示自己；補圖前顯示無圖色標，補圖後顯示各自頭像，砲台等共用技能仍正確。此缺陷是視覺驗收未過，不把自動流程PASS當全部頭像關係PASS。內容作者依範圍不改核心，整合者修復後重跑瀏覽器及release。

```typescript
import fs from 'node:fs';
import assert from 'node:assert/strict';
import {assemble} from '/workspace/20261020/src/content/loader.ts';
import {newSave,addCharacter,stats} from '/workspace/20261020/src/domain/model.ts';
import {apply,settle,summonCandidates} from '/workspace/20261020/src/domain/economy.ts';
import {Combat} from '/workspace/20261020/src/domain/combat.ts';
import {checkSave} from '/workspace/20261020/src/platform/storage.ts';
import {skillMechanic} from '/workspace/20261020/src/content/descriptions.ts';
const base=JSON.parse(fs.readFileSync('content/core/pack.json','utf8'));
const ext=JSON.parse(fs.readFileSync('content/extensions/solar10.json','utf8'));
const p=assemble(base,[ext]);
assert.equal(p.characters.length,16);assert.deepEqual(p.characters.slice(0,6),base.characters);
checkSave(newSave(base,0),p);
const rows=[];
for(const c of ext.definitions.characters){
 const pool=summonCandidates(p,c.recipe);assert(pool.some(v=>v.character.id===c.id));
 let obtained=false;
 for(let seed=1;seed<=100 && !obtained;seed++){
  let s=newSave(p,0);s.rng=Math.imul(seed,2654435761)>>>0;s.wallet=Object.fromEntries(p.resources.map(r=>[r.id,5000]));
  s=apply(p,s,{type:'summon',recipe:c.recipe,count:1},0,'summon');
  const expected=s.jobs[0].payload;
  for(let i=0;i<4;i++)assert.equal(s.wallet[p.resources[i].id],5000-c.recipe[i]);
  settle(p,s,p.rules.summonSeconds*1000+1);assert(s.roster.some(o=>o.definition===expected));
  obtained=s.roster.some(o=>o.definition===c.id);checkSave(s,p);
 }
 assert(obtained,c.name+' must actually summon');
 const s=newSave(p,0);const o=addCharacter(p,s,c.id)!;s.team=[o.id];s.x=610;s.y=1530;
 const b=new Combat(p,s,'m1:haven');const u=b.allies[0];const sk=b.skillFor(u)!;
 assert(b.cast(u.id));assert.equal(u.skillCooldown,sk.cooldown);assert(!b.cast(u.id));
 if(sk.effect==='fortify'||sk.effect==='accelerate'){assert.equal(u.buff,sk.effect);assert.equal(u.power,sk.power)}
 if(sk.effect==='shield')assert.equal(u.shield,sk.power);
 if(sk.effect==='weaken')assert(b.units.filter(v=>v.side==='enemy').every(v=>v.buff==='weaken'&&v.power===sk.power));
 if(sk.effect==='barrage'){assert.equal(b.bullets.length,12);assert.equal(b.bullets[0].attack,c.attack*sk.power)}
 if(sk.effect==='turret')assert(b.units.some(v=>v.turretUntil===sk.duration&&v.attack===c.attack*sk.power));
 const max=stats(p,{...o,level:p.rules.maxLevel,resonance:5});assert.equal(max.hp,Math.round(c.hp*(1+(p.rules.maxLevel-1)*.08)*1.1));
 rows.push({name:c.name,id:c.id,rank:pool.findIndex(v=>v.character.id===c.id)+1,skill:sk.id,mechanic:skillMechanic(sk),cooldown:sk.cooldown});
}
function fight(candidate:string,planet:string){
 const s=newSave(p,0);s.roster=[];s.team=[];
 const c=p.characters.find(v=>v.id===candidate)!;
 const ids=(c.role.startsWith('前鋒')||candidate==='m1:earth')?[candidate,'m1:mercury','m1:venus']:['m1:earth',candidate,'m1:venus'];
 for(const id of ids){const o=addCharacter(p,s,id)!;o.level=2;o.hp=stats(p,o).hp;s.team.push(o.id)}
 const v=p.planets.find(v=>v.id===planet)!;s.x=v.x-v.engageRadius+30;s.y=v.y;s.encounter={planet,seed:77};
 const b=new Combat(p,s,planet);b.autoSkills=true;
 for(let tick=0;tick<12000&&!b.ended;tick++){
 const e=b.units.filter(u=>u.side==='enemy'&&u.hp>0).sort((a,b2)=>Math.hypot(a.x-b.center.x,a.y-b.center.y)-Math.hypot(b2.x-b.center.x,b2.y-b.center.y))[0];
 if(e)b.destination={x:e.x-240+Math.cos(tick/80)*90,y:e.y+Math.sin(tick/80)*130};
 b.step(1/60);
 }
 assert(b.victory,c.name+' '+v.name);assert.equal(b.waveGroups.length,1);
 return {candidate:c.name,planet:v.name,seconds:+b.time.toFixed(1),alive:b.allies.filter(u=>u.hp>0).length};
}
const encounters=[];
for(const c of [...ext.definitions.characters,p.characters.find(c=>c.id==='m1:earth')!,p.characters.find(c=>c.id==='m1:mercury')!])for(const v of p.planets)encounters.push(fight(c.id,v.id));
fs.mkdirSync('.artifacts/solar10',{recursive:true});fs.writeFileSync('.artifacts/solar10/verification.json',JSON.stringify({rows,encounters},null,2));
fs.writeFileSync('/tmp/solar10-pack.json',JSON.stringify(p));
console.log(JSON.stringify({status:'PASS',characters:rows.length,encounters},null,2));
```
